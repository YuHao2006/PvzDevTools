cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PvzDat64Converter LANGUAGES CXX)
    include(CTest)
endif()
if(WIN32)
    add_executable(pvz_dat64_converter ConverterLauncher.cpp)
else()
    add_executable(pvz_dat64_converter main.cpp LevelSaveConverter.cpp)
endif()
target_compile_features(pvz_dat64_converter PRIVATE cxx_std_23)
target_include_directories(pvz_dat64_converter PRIVATE ../contracts)
if(MSVC)
    set_property(TARGET pvz_dat64_converter PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(pvz_dat64_converter PRIVATE /utf-8 /W4 /permissive-)
    target_compile_definitions(pvz_dat64_converter PRIVATE _SILENCE_CXX20_U8PATH_DEPRECATION_WARNING _WIN32_WINNT=0x0601 WINVER=0x0601)
endif()
install(TARGETS pvz_dat64_converter RUNTIME DESTINATION .)

if(BUILD_TESTING AND PVZ_TEST_FIXTURE_DIR)
    foreach(case IN ITEMS V4Conversion ParticleImageVariant ReanimationAppearance)
        string(TOLOWER "${case}" name)
        add_executable(test_${name} "${case}Tests.cpp")
        target_compile_features(test_${name} PRIVATE cxx_std_23)
        target_include_directories(test_${name} PRIVATE ../contracts)
        if(MSVC)
            set_property(TARGET test_${name} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_compile_options(test_${name} PRIVATE /utf-8)
        endif()
        if(case STREQUAL "V4Conversion")
            set(fixture game3_13.dat)
        else()
            set(fixture game3_58.dat64)
        endif()
        add_test(NAME ${name} COMMAND test_${name} "${PVZ_TEST_FIXTURE_DIR}/${fixture}")
    endforeach()
endif()

add_executable(pvz-save-repair repair_main.cpp LevelSaveConverter.cpp)

    target_compile_options(pvz-save-repair PRIVATE /utf-8 /W4 /permissive-)
    target_compile_definitions(pvz-save-repair PRIVATE _SILENCE_CXX20_U8PATH_DEPRECATION_WARNING _WIN32_WINNT=0x0601 WINVER=0x0601)

# Standalone cold-path library. No game, renderer, audio or resource dependencies.
add_library(pvz_dfu SHARED DataFixerUpper.cpp LevelSaveConverter.cpp)
target_compile_features(pvz_dfu PRIVATE cxx_std_23)
target_include_directories(pvz_dfu PRIVATE ../contracts)
set_target_properties(pvz_dfu PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
    target_link_options(pvz_dfu PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(MSVC)
    set_property(TARGET pvz_dfu PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(pvz_dfu PRIVATE /utf-8)
endif()

if(WIN32)
    target_sources(pvz_dfu PRIVATE main.cpp)
    set_source_files_properties(main.cpp PROPERTIES COMPILE_DEFINITIONS "main=PvzDfuCliEntry")
    target_compile_definitions(pvz_dfu PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601)
    add_dependencies(pvz_dat64_converter pvz_dfu)
    install(TARGETS pvz_dfu RUNTIME DESTINATION .)
endif()
